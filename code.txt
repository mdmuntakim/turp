User: write a recursive factorial in C
Turp:
```c
#include <stdio.h>

int fact(int n) {
    if (n <= 1) return 1;
    return n * fact(n - 1);
}

    printf("%d\n", fact(5));
    return 0;
}
```

User: write a C program for bubble sort
Turp:
```c
#include <stdio.h>

int main(void) {
    int a[] = {5, 1, 4, 2, 3};
    int n = 5;
    for (int i = 0; i < n; i++)
        for (int j = 0; j < n - 1; j++)
            if (a[j] > a[j + 1]) {
                int t = a[j]; a[j] = a[j + 1]; a[j + 1] = t;
            }
    for (int i = 0; i < n; i++) printf("%d ", a[i]);
    return 0;
}
```

User: write a C program for binary search
Turp:
```c
#include <stdio.h>

int main(void) {
    int a[] = {1, 3, 5, 7, 9};
    int n = 5, key = 7, lo = 0, hi = n - 1, pos = -1;
    while (lo <= hi) {
        int mid = (lo + hi) / 2;
        if (a[mid] == key) { pos = mid; break; }
        if (a[mid] < key) lo = mid + 1;
        else hi = mid - 1;
    }
    printf("%d\n", pos);
    return 0;
}
```
